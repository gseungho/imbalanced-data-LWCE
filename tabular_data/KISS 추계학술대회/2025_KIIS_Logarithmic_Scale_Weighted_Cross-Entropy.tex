\begin{frame}{Latent Majority Bias in Imbalanced Data}

\vspace{0.03\textwidth} % 상단 여백 조절

\begin{columns}[T] % 'T' (Top-aligned) 옵션

  \begin{itemize}

% Page 4: Common Approaches to Class Imbalance
\begin{frame}{Common Approaches to Class Imbalance}

\vspace{0.03\textwidth} % 전체적인 상단 여백 살짝 조절

% --- 상단: 두 가지 주요 접근법 ---
\begin{columns}[T] % 'T' (top-aligned) 옵션

  % --- 1. Data Resampling ---
  \column{0.5\textwidth}
  \bblock{1. Data Resampling}
    \begin{itemize}
        \item \textbf{What it is:} Modifies the training set distribution to achieve a better balance.
        \item \textbf{Examples:}
        \begin{itemize}
            \item Oversampling (SMOTE, ADASYN)
            \item Undersampling (Tomek Links, ENN)
        \end{itemize}
    \end{itemize}
  \eblock

  % --- 2. Cost-Sensitive Learning ---
  \column{0.5\textwidth}
  \bblock{2. Cost-Sensitive Learning}
    \begin{itemize}
        \item \textbf{What it is:} Adjusts the algorithm to assign a higher penalty for misclassifying minority samples.
        \item \textbf{Examples:}
        \begin{itemize}
            \item Weighted Cross-Entropy (WCE)
            \item Focal Loss, Class-Balanced Loss
        \end{itemize}
    \end{itemize}
  \eblock

\end{columns}

\vspace{0.05\textwidth}

% --- 하단: 왜 Cost-Sensitive가 필요한가?
\begin{columns}[T]
  
  % --- CE 수식 ---
  \column{0.4\textwidth}
    \textbf{Standard Cross-Entropy}:
    $$ \mathcal{L}_{CE} = - \sum_{i=1}^{N} y_i \log(p_i) $$

  \column{0.6\textwidth}
    \textbf{Problem:} The total loss is \alert{dominated} by the majority class, and the gradient signal from the minority class is overwhelmed.
    
    \vspace{0.3cm}
    $\to$ \textit{Focus on \textbf{Cost-Sensitive Learning}.}

\end{columns}

\end{frame}

% Page 5: Existing Methods (I) - Inverse Weighting
\begin{frame}{Existing Methods (I): Inverse Weighting Schemes}

\vspace{0.03\textwidth}
The simplest cost-sensitive method is to assign a weight $w_i$ to the standard CE loss:
$$ \mathcal{L}_{WCE} = - \sum_{i=1}^{N} \alert{w_i} \cdot y_i \log(p_i) $$


\begin{block}{Weighted Cross-Entropy (WCE) \& Inverse Sqrt WCE}
    \begin{itemize}
        \item \textbf{Core Idea:} Weights are inversely proportional to class frequency, sometimes softened.
        \item \textbf{Weight Formula} ($n_i$: samples in class $i$)\textbf{:}
              $$ w_i \propto \frac{1}{n_i} \quad \text{or} \quad w_i \propto \frac{1}{\sqrt{n_i}} $$
    \end{itemize}
\end{block}

\end{frame}
%------------------------------------------------------------


% Page 6: Existing Methods (II) - Advanced Losses
\begin{frame}{Existing Methods (II): Advanced Losses}

\vspace{0.03\textwidth}

\begin{columns}[T]

  % --- 1. Focal Loss ---
  \column{0.5\textwidth}
  \bblock{1. Focal Loss}
    \begin{itemize}
        \item \textbf{Core Idea:} Focuses on \alert{sample difficulty} instead of class frequency.
        \item Down-weights the loss of "easy" (often majority) samples.
        \item \textbf{Formula} ($p_t$: prob. of true class)\textbf{:}
              $$ \mathcal{L}_{FL} = - \alpha_t (1 - p_t)^{\gamma} \log(p_t) $$
        \item \textbf{Key Hyperparameter:}
              \begin{itemize}
                  \item $\gamma$ controls the focusing rate.
              \end{itemize}
    \end{itemize}
  \eblock

  % --- 2. Class-Balanced (CB) Loss ---
  \column{0.5\textwidth}
  \bblock{2. Class-Balanced (CB) Loss}
    \begin{itemize}
        \item \textbf{Core Idea:} A principled re-weighting based on the "effective number of samples".
        \item \textbf{Weight Formula ($N_i(n_i, \beta)$):}
              $$ w_i = \frac{1 - \beta}{1 - \beta^{n_i}} $$
              $$ \mathcal{L}_{CB} = - \sum w_i \cdot \mathcal{L}_{CE}(y_i, p_i) $$
        \item \textbf{Key Hyperparameter:}
              \begin{itemize}
                  \item $\beta$ controls the curve.
              \end{itemize}
    \end{itemize}
  \eblock

\end{columns}
\end{frame}

% Page 8: Proposed Methodology (I) - The Core Idea
\begin{frame}{Proposed Methodology (I): The Core Idea}

\vspace{0.03\textwidth}

\begin{columns}[T]

  % --- 왼쪽: 문제점 ---
  \column{0.4\textwidth}
  \begin{block}{Motivation: Instability of WCE}
    Standard WCE weight: $$ w_i = \frac{N_{total}}{n_i} \propto \frac{1}{n_i} $$
    
    \begin{alertblock}{Critical Issue}
    Severe imbalance ($n_i \to \text{small}$) causes $w_i$ to \alert{explode}.
    \begin{itemize}
        \item Numerical instability.
        \item Severe overfitting to minority
    \end{itemize}
    \end{alertblock}
  \end{block}

  % --- 오른쪽: 해결책 ---
  \column{0.6\textwidth}
  \bblock{Solution: Log-Weighted CE (LWCE)}
    We use log-scaling to \alert{tame} the weights:
    \vspace{-0.02\textwidth}
    $$ w_i \propto \frac{1}{\log(n_i + 1)} $$
    
    \vspace{-0.03\textwidth}
    \centering
    \includegraphics[width=0.9\textwidth, height=0.55\textheight]{wce_vs_lwce_relative_v2.pdf}
  \eblock
  
\end{columns}
\end{frame}
%------------------------------------------------------------

% Page 9: Proposed Methodology (II) - Generalization (PLWCE)
\begin{frame}{Proposed Methodology (II): Power-Scaled LWCE (PLWCE)}

\vspace{0.06\textwidth}

% --- 상단: PLWCE 수식 ---
\bblock{Generalization: Power-Scaled Adaptive CE (PLWCE)}
    We introduce $\alpha$ to flexibly control the \alert{focusing strength}:
    $$ w_i \propto \frac{1}{(\log(n_i + 1))^{\alert{\alpha}}} $$
\eblock

\begin{columns}[T]

  % --- 왼쪽: 알파의 역할 (표로 정리) ---
  \column{0.45\textwidth}
  \begin{table}
      \centering
      \begin{tabular}{cl}
          \textbf{Alpha ($\alpha$)} & \textbf{Role \& Effect} \\
          \midrule
          $\alpha = 1.0$ & Balanced LWCE \\
          \addlinespace
          $\alpha > 1.0$ & \alert{Aggressive} focus on minority \\
          \addlinespace
          $\alpha < 1.0$ & Conservative weighting \\
          \bottomrule
      \end{tabular}
      \caption{Role of the $\alpha$ parameter}
  \end{table}

  % --- 오른쪽: 알파 비교 그래프 ---
  \column{0.55\textwidth}
  \vspace{-0.02\textwidth}
  \centering
  \includegraphics[width=0.9\textwidth, height=0.55\textheight]{plwce_alpha_relative_zoomed.pdf}
\end{columns}

\end{frame}

% Page 10: Proposed Methodology (III): Power-Scaled WCE (PS-WCE)
\begin{comment}
\begin{frame}{Proposed Methodology (III): Power-Scaled WCE (PS-WCE)}

\vspace{0.06\textwidth}

% --- 상단: PS-WCE 수식 ---
\bblock{Generalization of Standard WCE}
    This approach generalizes standard WCE by adding an exponent $\alpha$:
    $$ w_i \propto \frac{1}{(n_i)^{\alert{\alpha}}} $$
\eblock

\begin{columns}[T]

  % --- 왼쪽: 알파의 역할 (표로 정리) ---
  \column{0.45\textwidth}
  \begin{table}
    \centering
    \begin{tabular}{cl}
        \textbf{Alpha ($\alpha$)} & \textbf{Role \& Effect} \\
        \midrule
        $\alpha = 1.0$ & Standard WCE (Explodes) \\
        \addlinespace
        $\alpha = 0.5$ & Sqrt WCE (Standard fix) \\
        \addlinespace
        $\alpha < 0.5$ & More Conservative (Flatter) \\
        \bottomrule
    \end{tabular}
    \caption{Role of the $\alpha$ parameter}
  \end{table}
  \vspace{0.2cm}
  \begin{alertblock}{Limitation}
      Unlike LWCE, this method still suffers from \alert{potential instability} as $n_i \to 0$, just at a different rate.
  \end{alertblock}

% Page 12: Experimental Setup (II) - Metrics & Optimization
\begin{frame}{Experimental Setup (II): Metrics & Optimization}

\vspace{0.03\textwidth}

\begin{columns}[T]

  % --- 왼쪽: 평가 지표 ---
  \column{0.5\textwidth}
  \bblock{Evaluation Metrics}
    Standard accuracy is misleading for imbalanced data. We focus on:
    \begin{itemize}
        \item \textbf{F1-Score:} Harmonic mean of precision and recall, averaged across classes.
        \item \textbf{PR-AUC:} Area under the Precision-Recall curve, highly sensitive to minority class performance.
    \end{itemize}
  \eblock

  % --- 오른쪽: 최적화 (Optuna) ---
  \column{0.5\textwidth}
  \bblock{Hyperparameter Optimization}
    We used \textbf{Optuna} for automated hyperparameter tuning.
    \begin{itemize}
        \item \textbf{Search Space:}
        \begin{itemize}
            \item \texttt{learning\_rate}, \texttt{n\_estimators}
            \item Regularization: \texttt{lambda} (L2), \texttt{alpha} (L1)
            \item Loss-specific: $\gamma$ (Focal), $\beta$ (CB), $\alpha$ (PS-WCE, PLWCE)
        \end{itemize}
        \item \textbf{Protocol:} 3-fold CV, 100 trials per dataset.
    \end{itemize}
  \eblock

\end{columns}

\end{frame}

\begin{comment}
\begin{frame}{Discussion (I): The Role of $\alpha$ as a Trade-off Tuner}

\vspace{0.02\textwidth}
\begin{block}{Observation from Results}
    \begin{itemize}
        \item Optimizing for F1-Score often resulted in $\alpha > 1.0$ (PLWCE).
        \item However, in several cases (e.g., Bank Marketing, Telco Churn), this led to a lower PR-AUC score compared to LWCE ($\alpha=1.0$).
    \end{itemize}
\end{block}

\vspace{-0.015\textwidth}
\bblock{Interpretation: The Precision-Recall Trade-off}
    The $\alpha$ parameter acts as a "Recall vs. Precision" tuner:
    \begin{itemize}
        \item \textbf{LWCE ($\alpha = 1.0$):} Provides a stable, balanced baseline with high \alert{Precision} (good for PR-AUC).
        \vfill
        \item \textbf{PLWCE ($\alpha > 1.0$):} \alert{Aggressively} focuses on the minority class, leading to higher \alert{Recall} (good for F1-Score), but can sometimes hurt Precision.
    \end{itemize}
    \vfill
    $\to$ \textit{This is not a flaw, but a feature: The choice of $\alpha$ allows specialization based on the problem's needs (e.g., recall in medical vs. precision in spam filtering).}
\eblock

\end{frame}
\end{comment}

% Page 17: Discussion - Why Tune Alpha?
\begin{frame}{Discussion (I): The Benefit of Tuning $\alpha$}

\vspace{0.02\textwidth}
\begin{block}{Observation from Results}
    \begin{itemize}
        \item The results show that LWCE ($\alpha=1.0$) is already a very strong and robust baseline, often outperforming SOTA methods.
        \item However, in 5 out of 11 datasets, PLWCE (tuning $\alpha > 1.0$) achieved an even higher F1-Score than LWCE.
    \end{itemize}
\end{block}

\vspace{-0.01\textwidth}
\bblock{Conclusion: Alpha as a "Performance Booster"}
    $\alpha$ acts as a data-specific "booster":
    \begin{itemize}
        \item LWCE ($\alpha=1.0$) is sufficient and provides top-tier performance.
        \item For some \alert{extremely imbalanced} datasets, tuning $\alpha > 1$ allows PLWCE to find a better optimum, boosting \textbf{both F1 and PR-AUC} scores.
    \end{itemize}
    \vfill
    $\to$ \textit{PLWCE generalizes LWCE, providing an optional tuning parameter for achieving the best possible performance in critical scenarios.}
\eblock

\end{frame}

\begin{frame}{Discussion (II): Difficulty of Isolating $\alpha$}

  \column{0.45\textwidth}
  \bblock{Challenge: Isolating $\alpha$ vs. IR}
    \begin{itemize}
        \item \textbf{Ideal Experiment:} Fix all model parameters (LR, L1/L2, etc.) across all 11 datasets, and tune only $\alpha$.
        \vfill
        \item \textbf{Problem:} A single set of fixed parameters failed. It was too sub-optimal for diverse datasets (e.g., \texttt{credit\_card\_fraud}), causing poor metrics and rendering the entire imbalanced data analysis meaningless under these conditions

\begin{frame}{Future Work (I): Model \& Weighting Expansion}

\vspace{0.03\textwidth}

\bblock{1. Application to Other Models}
    \begin{itemize}
        \item \textbf{Tree-based Models:} Validate LWCE/PLWCE on non-linear `gbtree` models (standard XGBoost, LightGBM).
        \vfill
        \item \textbf{Deep Learning:} Apply to CNNs/Transformers for highly imbalanced tasks like object detection or medical segmentation.
    \end{itemize}
\eblock

\vspace{0.01\textwidth}

\bblock{2. Combining with Dynamic Sample Weighting}
    \begin{itemize}
        \item Our method (PLWCE) uses \alert{static, class-level} weights.
        \vfill
        \item \textbf{Hypothesis:} We can combine this with \alert{dynamic, sample-level} methods (like Focal Loss or GHM) that focus on "easy" vs. "hard" samples.
        \vfill
        \item \textbf{Goal:} Investigate a hybrid loss that benefits from both stable class-based re-weighting (our $\alpha$) and dynamic sample-level focus (like Focal's $\gamma$).
    \end{itemize}
\eblock

\end{frame}

% Page 20: Future Work (II)
\begin{frame}{Future Work (II): Automated Alpha ($\alpha$) Tuning}

\vspace{0.05\textwidth}

\bblock{3. Automated Alpha ($\alpha$) Tuning}
    \begin{itemize}
        \item As discussed (P18), isolating $\alpha$'s effect is challenging due to other interacting hyperparameters.
        \vfill
        \item \textbf{Controlled Experiment:}
        \begin{itemize}
            \item Use a large dataset (e.g., \texttt{credit\_card\_fraud}).
            \item Fix minority samples and \alert{sub-sample the majority class} to create 5+ datasets with controlled IRs (e.g., 5:1, 10:1, 100:1).
        \end{itemize}
        \vfill
        \item \textbf{Goal:} Find a clear correlation between \alert{IR} and the \alert{optimal $\alpha$} by running alpha-only tuning on these controlled datasets.
        \vfill
        \item \textbf{Ultimate Aim:} Develop a heuristic or formula to set $\alpha$ \alert{automatically}, eliminating the need for expensive trial-and-error tuning.
    \end{itemize}
\eblock

% Page 21: Conclusion
\begin{frame}{Conclusion}

\vspace{0.03\textwidth}
\bblock{Key Contributions}
    \begin{enumerate}
        \item \textbf{Identified Instability:} We highlighted the numerical instability of standard WCE ($1/n_i$) in extreme imbalance scenarios.
        \vfill
        \item \textbf{Proposed LWCE:} We proposed \textbf{LWCE} ($1/\log(n_i+1)$) as a simple, stable, and parameter-free solution that fundamentally tames weight explosion.
        \vfill
        \item \textbf{Proposed PLWCE:} We generalized this into \textbf{PLWCE} ($1/(\log n_i)^\alpha$), demonstrating how $\alpha$ acts as a performance booster for extremely imbalanced datasets.
        \vfill
        \item \textbf{Validated Performance:} Our methods proved \alert{highly robust and competitive}, consistently ranking in the Top 2 across 11 diverse datasets against strong baselines.
    \end{enumerate}
\eblock
